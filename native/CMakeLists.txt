cmake_minimum_required(VERSION 3.21)
project(Xnounce VERSION 0.4.0 LANGUAGES C CXX)

# -----------------------------------------------------------------------------
# Xnounce: Fenix-style Virtual Purser cabin announcements for the ToLiss Airbus family.
# Output layout (X-Plane "fat plugin"):
#   <build>/Xnounce/{lin_x64,win_x64,mac_x64}/Xnounce.xpl
# -----------------------------------------------------------------------------

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()
if(APPLE)
    set(CMAKE_OSX_ARCHITECTURES "x86_64;arm64" CACHE STRING "" FORCE)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "" FORCE)
endif()

option(XN_STATIC_RUNTIME "Linux: link libstdc++/libgcc statically (release builds)" OFF)
option(XN_PANEL_GRAPHICS "Render via Panel Graphics on XP 12.4.4+ (dynamic, OpenGL fallback)" ON)

# ---- third-party (pinned) -----------------------------------------------------
include(FetchContent)
set(FETCHCONTENT_QUIET ON)
FetchContent_Declare(xpsdk
    URL https://developer.x-plane.com/wp-content/plugins/code-sample-generation/sdk_zip_files/XPSDK430.zip
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_Declare(imgui
    GIT_REPOSITORY https://github.com/ocornut/imgui.git
    GIT_TAG v1.92.9b
    GIT_SHALLOW TRUE)
FetchContent_Declare(imgwindow
    GIT_REPOSITORY https://github.com/slgoldberg/ImgWindow.git
    GIT_TAG ef979d57cb86e50cb7e98c1036cd9fd4805eccc7)   # v2.2.2: XPLM 4.4b3+ Panel Graphics (global coords)
# libogg/libvorbis are compiled from source below (decoder only); SOURCE_SUBDIR points at a folder that
# doesn't exist so FetchContent only downloads them and never runs their own CMake.
FetchContent_Declare(ogg
    GIT_REPOSITORY https://github.com/xiph/ogg.git
    GIT_TAG v1.3.6
    GIT_SHALLOW TRUE
    SOURCE_SUBDIR _xn_no_cmake)
FetchContent_Declare(vorbis
    GIT_REPOSITORY https://github.com/xiph/vorbis.git
    GIT_TAG v1.3.7
    GIT_SHALLOW TRUE
    SOURCE_SUBDIR _xn_no_cmake)
FetchContent_Declare(doctest
    GIT_REPOSITORY https://github.com/doctest/doctest.git
    GIT_TAG v2.4.11
    GIT_SHALLOW TRUE
    SOURCE_SUBDIR _xn_no_cmake)
FetchContent_MakeAvailable(xpsdk imgui imgwindow ogg vorbis doctest)

set(XPSDK_DIR ${xpsdk_SOURCE_DIR})
if(EXISTS ${XPSDK_DIR}/SDK)
    set(XPSDK_DIR ${XPSDK_DIR}/SDK)
endif()
set(XN_XPLM_DEFS XPLM200=1 XPLM210=1 XPLM300=1 XPLM301=1 XPLM303=1 XPLM400=1 XPLM410=1 XPLM420=1 XPLM430=1)

    ${vorbis_SOURCE_DIR}/lib/bitrate.c

    ${vorbis_SOURCE_DIR}/lib/codebook.c
    ${vorbis_SOURCE_DIR}/lib/envelope.c
    ${vorbis_SOURCE_DIR}/lib/floor0.c
    ${vorbis_SOURCE_DIR}/lib/floor1.c
    ${vorbis_SOURCE_DIR}/lib/info.c
    ${vorbis_SOURCE_DIR}/lib/lookup.c
    ${vorbis_SOURCE_DIR}/lib/lpc.c
    ${vorbis_SOURCE_DIR}/lib/lsp.c
    ${vorbis_SOURCE_DIR}/lib/mapping0.c

if(MSVC)
    target_compile_definitions(xn_vorbis PRIVATE _CRT_SECURE_NO_WARNINGS _USE_MATH_DEFINES)
    target_compile_options(xn_vorbis PRIVATE /W0)
else()
    target_compile_options(xn_vorbis PRIVATE -w)
    if(NOT APPLE)
        target_link_libraries(xn_vorbis PUBLIC m)
    endif()

# ---- sources ------------------------------------------------------------------
# Pure C++ (no XPLM, no ImGui): linked into the plugin and the unit tests.
# native/data/airlines.tsv is embedded as a string literal (one literal: keep it under MSVC's 16 KB limit).
file(READ ${CMAKE_CURRENT_SOURCE_DIR}/data/airlines.tsv XN_AIRLINES_TSV)
string(LENGTH "${XN_AIRLINES_TSV}" XN_AIRLINES_LEN)
if(XN_AIRLINES_LEN GREATER 16000)
    message(FATAL_ERROR "data/airlines.tsv is ${XN_AIRLINES_LEN} bytes; split the embedding in airlines_data.cpp.in")
endif()
configure_file(src/core/airlines_data.cpp.in ${CMAKE_BINARY_DIR}/generated/airlines_data.cpp @ONLY)

set(XN_CORE_SOURCES
    src/core/aircraft.cpp
    src/core/airlines.cpp
    src/core/announcement.cpp
    src/core/config.cpp
    src/core/guard.cpp
    src/core/log.cpp
    src/core/pack_library.cpp
    src/core/pack_resolver.cpp
    src/core/purser.cpp
    src/core/recorder.cpp
    src/core/refuel.cpp
    src/core/selector.cpp
    src/core/simbrief.cpp
    src/core/sound_file.cpp
    src/core/strings.cpp
    ${CMAKE_BINARY_DIR}/generated/airlines_data.cpp
)
# Audio without XPLM (decoder, decode cache, chime, hearing model): plugin and unit tests.
set(XN_AUDIO_SOURCES
    src/audio/chime.cpp
    src/audio/decode_cache.cpp
    src/audio/decoder.cpp
    src/audio/hearing.cpp
    src/audio/muffle.cpp

# Panel content (ImGui only, no XPLM, no ImgWindow): linked into the plugin and the headless UI test.
set(XN_UI_SOURCES
    src/ui/theme.cpp
    src/ui/fonts_b612.cpp
    src/ui/widgets.cpp
    src/ui/panel.cpp
    src/ui/tab_status.cpp

# X-Plane glue.
set(XN_XP_SOURCES
    src/plugin.cpp

    src/sim/detect.cpp
    src/sim/simbrief_source.cpp

    src/audio/player.cpp

set(THIRD_PARTY_SOURCES
    ${imgui_SOURCE_DIR}/imgui.cpp
    ${imgui_SOURCE_DIR}/imgui_draw.cpp
    ${imgui_SOURCE_DIR}/imgui_tables.cpp
    ${imgui_SOURCE_DIR}/imgui_widgets.cpp

add_library(Xnounce SHARED ${XN_CORE_SOURCES} ${XN_AUDIO_SOURCES} ${XN_UI_SOURCES} ${XN_XP_SOURCES} ${THIRD_PARTY_SOURCES})

target_include_directories(Xnounce PRIVATE
    src
    ${imgui_SOURCE_DIR}
    ${imgwindow_SOURCE_DIR}
    fonts
    ${XPSDK_DIR}/CHeaders/XPLM
    ${XPSDK_DIR}/CHeaders/Widgets)
target_link_libraries(Xnounce PRIVATE xn_vorbis)

target_compile_definitions(Xnounce PRIVATE ${XN_XPLM_DEFS} XN_VERSION="${PROJECT_VERSION}")
if(XN_PANEL_GRAPHICS)
    target_compile_definitions(Xnounce PRIVATE IMGWINDOW_USE_PANEL_GRAPHICS)
endif()

if(WIN32)
    target_compile_definitions(Xnounce PRIVATE IBM=1 WIN32_LEAN_AND_MEAN NOMINMAX _USE_MATH_DEFINES)
    target_link_libraries(Xnounce PRIVATE ${XPSDK_DIR}/Libraries/Win/XPLM_64.lib opengl32)
    # MSVC has no <GL/glext.h>; ImgWindow needs only GL 1.1, so an empty shim satisfies its include.
    target_include_directories(Xnounce PRIVATE compat/win)
    set(XN_PLATFORM_DIR win_x64)
elseif(APPLE)
    target_compile_definitions(Xnounce PRIVATE APL=1 GL_SILENCE_DEPRECATION)
    find_library(XPLM_FRAMEWORK XPLM PATHS ${XPSDK_DIR}/Libraries/Mac NO_DEFAULT_PATH)
    find_library(OPENGL_FRAMEWORK OpenGL)
    target_link_libraries(Xnounce PRIVATE ${XPLM_FRAMEWORK} ${OPENGL_FRAMEWORK})
    set(XN_PLATFORM_DIR mac_x64)
else()
    target_compile_definitions(Xnounce PRIVATE LIN=1)
    find_package(OpenGL REQUIRED)
    find_package(Threads REQUIRED)
    target_link_libraries(Xnounce PRIVATE OpenGL::GL Threads::Threads)
    # X-Plane resolves XPLM symbols at load time; only XPlugin* are exported.
    target_link_options(Xnounce PRIVATE -Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map
        $<$<CONFIG:Release>:-s>)
    # Release builds: static C++ runtime, since users' distros often ship an older libstdc++ than the
    # build machine (needs the libstdc++ static archive; CI turns this on).
    if(XN_STATIC_RUNTIME)
        target_link_options(Xnounce PRIVATE -static-libstdc++ -static-libgcc)
    endif()
    set(XN_PLATFORM_DIR lin_x64)
endif()

if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(Xnounce PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall -Wextra>)
    set_source_files_properties(${THIRD_PARTY_SOURCES} PROPERTIES COMPILE_OPTIONS "-w")
endif()

set_target_properties(Xnounce PROPERTIES
    PREFIX ""
    OUTPUT_NAME Xnounce
    SUFFIX ".xpl"
    LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/Xnounce/${XN_PLATFORM_DIR}
    RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/Xnounce/${XN_PLATFORM_DIR})

# ---- unit tests (pure C++, no X-Plane) -------------------------------------------------------------
add_executable(xn_tests EXCLUDE_FROM_ALL tests/test_main.cpp tests/test_core.cpp tests/test_packs.cpp tests/test_purser.cpp tests/test_audio.cpp
    tests/test_toliss.cpp
    ${XN_CORE_SOURCES} ${XN_AUDIO_SOURCES})
target_include_directories(xn_tests PRIVATE src ${doctest_SOURCE_DIR})
find_package(Threads REQUIRED)
target_link_libraries(xn_tests PRIVATE Threads::Threads xn_vorbis)
# DOCTEST_CONFIG_USE_STD_HEADERS: doctest otherwise forward-declares std::ostream, and MSVC then can't
# stringify std::string_view in a failed CHECK (C2027 "undefined type basic_ostream").
target_compile_definitions(xn_tests PRIVATE XN_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}" DOCTEST_CONFIG_USE_STD_HEADERS)

# ---- headless UI test: every tab, many frames, ImGui asserts on (null backend) ----------------------
add_executable(xn_ui_tests EXCLUDE_FROM_ALL tests/test_ui.cpp ${XN_CORE_SOURCES} ${XN_UI_SOURCES}
    ${imgui_SOURCE_DIR}/backends/imgui_impl_null.cpp
    ${imgui_SOURCE_DIR}/imgui.cpp
    ${imgui_SOURCE_DIR}/imgui_draw.cpp
    ${imgui_SOURCE_DIR}/imgui_tables.cpp
    ${imgui_SOURCE_DIR}/imgui_widgets.cpp)
target_include_directories(xn_ui_tests PRIVATE src fonts ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
# ImGui asserts stay on in Release on every compiler (MSVC takes -U too and warns D9025 about overriding /DNDEBUG).
target_compile_options(xn_ui_tests PRIVATE -UNDEBUG)
